-- Drop and recreate pets table (optional if already exists)
drop table if exists public.pets cascade;

create table public.pets (
  id bigint generated by default as identity primary key,
  created_at timestamp with time zone not null default now(),
  name text not null,
  type text not null,
  age integer,
  image_url text
);

-- Enable RLS
alter table public.pets enable row level security;

-- Allow public (even anon) to read pets
create policy "Allow anyone to read pets"
on public.pets
for select
using (true);
